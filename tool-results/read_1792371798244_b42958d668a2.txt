  1523→function createNode(data) {
  1524→  const id = data.id || ('n' + (nextId++));
  1525→  const node = {
  1526→    id,
  1527→    name: data.name,
  1528→    type: data.type || 'folder',
  1529→    parentId: data.parentId || null,
  1530→    children: (data.type === undefined || data.type === 'folder') ? (data.children || []) : null,
  1531→    size: data.size || 0,
  1532→    modified: data.modified || randomDate(),
  1533→    meta: data.meta || {},
  1534→    expanded: data.expanded || false,
  1535→  };
  1536→  fsNodes.set(id, node);
  1537→  return node;
  1538→}

  1540→function randomDate() {
  1541→  const days = Math.floor(Math.random() * 120);
  1542→  const d = new Date(Date.now() - days * 86400000 - Math.random() * 86400000);
  1543→  return d;
  1544→}

  1546→function getChildByName(parentId, name) {
  1547→  const parent = fsNodes.get(parentId);
  1548→  if (!parent || !parent.children) return null;
  1549→  for (const cid of parent.children) {
  1550→    const c = fsNodes.get(cid);
  1551→    if (c && c.name === name) return c;
  1552→  }
  1553→  return null;
  1554→}
  1555→
  1556→function getPath(nodeId) {
  1557→  const parts = [];
  1558→  let n = fsNodes.get(nodeId);
  1559→  while (n) {
  1560→    parts.unshift({ id: n.id, name: n.name });
  1561→    n = n.parentId ? fsNodes.get(n.parentId) : null;
  1562→  }
  1563→  return parts;
  1564→}
  1565→
  1566→function findNodeByPath(parts) {
  1567→  let cur = fsNodes.get('root');
  1568→  for (const p of parts) {
  1569→    if (!cur || !cur.children) return null;
  1570→    let found = null;
  1571→    for (const cid of cur.children) {
  1572→      const c = fsNodes.get(cid);
  1573→      if (c && c.name === p) { found = c; break; }
  1574→    }
  1575→    if (!found) return null;
  1576→    cur = found;
  1577→  }
  1578→  return cur ? cur.id : null;
  1579→}

  1581→function isAncestor(ancestorId, nodeId) {
  1582→  let n = fsNodes.get(nodeId);
  1583→  while (n && n.parentId) {
  1584→    if (n.parentId === ancestorId) return true;
  1585→    n = fsNodes.get(n.parentId);
  1586→  }
  1587→  return false;
  1588→}

  1972→  html += `<span class="tree-arrow ${node.expanded ? 'expanded' : ''} ${hasChildren ? '' : 'empty'}" data-node-id="${nodeId}">
  1973→    <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><polyline points="9 18 15 12 9 6"></polyline></svg>
  1974→  </span>`;
  1975→  html += `<i data-lucide="${node.expanded ? 'folder-open' : 'folder'}" class="tree-icon"></i>`;
  1976→  html += `<span class="tree-label">${escapeHtml(node.name)}</span>`;
  1977→  html += `</div>`;
  1978→
  1979→  if (hasChildren && node.expanded) {

  1981→    for (const cid of node.children) {
  1982→      const child = fsNodes.get(cid);
  1983→      if (child && child.type === 'folder') {
  1984→        html += renderTreeNode(cid, depth + 1, activeNodeId);
  1985→      }
  1986→    }
  1987→    html += `</div>`;
  1988→  }
  1989→
  1990→  html += `</div>`;
  1991→  return html;
  1992→}
