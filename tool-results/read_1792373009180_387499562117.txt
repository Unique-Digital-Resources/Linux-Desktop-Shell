     1→# FileExplorer.html - Analysis
     2→
     3→## Overview
     4→
     5→**FileExplorer.html** is a self-contained, single-file web app that emulates a
     6→modern desktop file manager inside a browser. Branded **"Aether — Files"**, it
     7→presents a light, Google Material You-inspired UI (off-white workspace, soft
     8→blue accent `#0B57D0`, rounded surfaces, subtle shadows) using the Inter font
     9→for sans text and JetBrains Mono for code/shortcuts. Icons are provided by the
    10→Lucide icon library (loaded from a CDN with a graceful fallback that draws a
    11→placeholder dot when `window.lucide` is unavailable).
    12→
    13→### Visual concept
    14→- Light theme with `--bg-workspace: #FBFAF8`, `--bg-tabbar: #F0EDE8`,
    15→  white sidebar, and `--accent: #0B57D0` (Google-blue) for selections,
    16→  active states, focus rings, and primary buttons.
    17→- Soft elevation via `--shadow-pop` / `--shadow-dialog`, generous use of
    18→  `var(--radius-*)` rounding, and a 100-260 ms cubic-bezier transition system.
    19→- Status colors: `--success` green, `--warning` amber, `--error` red,
    20→  `--info` blue. Storage bars switch color thresholds above 75 % / 90 %.
    21→
    22→### Main UI regions
    23→1. **Sidebar** (`aside.sidebar`): brand header ("A" gradient tile + "Aether Files"
    24→   wordmark), a scrollable nav area (Favorites, Places, Directory Tree, Storage
    25→   & Network sections), and a right-edge resize handle.
    26→2. **Tabstrip** (`.tab-strip`): browser-style horizontal tabs with per-tab
    27→   icon, label, close button, an active-tab accent bar (`::after`), and a
    28→   trailing "+" new-tab button. Tabs are right-clickable for a context menu.
    29→3. **Toolbar** (`.toolbar`): back/forward/up/refresh `btn-group`; a
    30→   `dir-bar-wrap` containing the breadcrumb trail plus dir-actions
    31→   (new folder, new file, paste); and a leading-icon search input.
    32→4. **Action bar** (`.action-bar`, auto-hides): copy / cut / rename / delete
    33→   `btn-group` + a more-actions button + a `.selection-info` count pill.
    34→5. **Content area** (`.content` > `.content-scroll`): renders either the
    35→   Dashboard widget view (Win11-inspired image gallery widget with vertical
    36→   masonry or horizontal scroll modes) or a file grid / list / compact list,
    37→   with empty-state placeholders for no results / empty folder.
    38→6. **Status bar** (`.statusbar`): item count + selection summary on the left;
    39→   a `.view-segmented` control (grid/list/compact), a sort button, a divider,
    40→   and a storage indicator on the right.
    41→7. **Dialogs & overlays**: generic modal dialog (rename / new folder / new file
    42→   / confirm-trash), a fixed `.properties-panel` for file details, a
    43→   `.toast-container` for transient notifications, and floating
    44→   `.context-menu` popups (file, background, sort, more, tab).

   132→function createNode(data) {
   133→  const id = data.id || ('n' + (nextId++));
   134→  const node = {
   135→    id,
   136→    name: data.name,
   137→    type: data.type || 'folder',
   138→    parentId: data.parentId || null,
   139→    children: (data.type === undefined || data.type === 'folder') ? (data.children || []) : null,
   140→    size: data.size || 0,
   141→    modified: data.modified || randomDate(),
   142→    meta: data.meta || {},
   143→    expanded: data.expanded || false,
   144→  };
   145→  fsNodes.set(id, node);
   146→  return node;
   147→}

   149→function randomDate() {
   150→  const days = Math.floor(Math.random() * 120);
   151→  const d = new Date(Date.now() - days * 86400000 - Math.random() * 86400000);
   152→  return d;
   153→}

   155→function getChildByName(parentId, name) {
   156→  const parent = fsNodes.get(parentId);
   157→  if (!parent || !parent.children) return null;
   158→  for (const cid of parent.children) {
   159→    const c = fsNodes.get(cid);
   160→    if (c && c.name === name) return c;
   161→  }
   162→  return null;
   163→}
   164→
   165→function getPath(nodeId) {
   166→  const parts = [];
   167→  let n = fsNodes.get(nodeId);
   168→  while (n) {
   169→    parts.unshift({ id: n.id, name: n.name });
   170→    n = n.parentId ? fsNodes.get(n.parentId) : null;
   171→  }
   172→  return parts;
   173→}
   174→
   175→function findNodeByPath(parts) {
   176→  let cur = fsNodes.get('root');
   177→  for (const p of parts) {
   178→    if (!cur || !cur.children) return null;
   179→    let found = null;
   180→    for (const cid of cur.children) {
   181→      const c = fsNodes.get(cid);
   182→      if (c && c.name === p) { found = c; break; }
   183→    }
   184→    if (!found) return null;
   185→    cur = found;
   186→  }
   187→  return cur ? cur.id : null;
   188→}

   190→function isAncestor(ancestorId, nodeId) {
   191→  let n = fsNodes.get(nodeId);
   192→  while (n && n.parentId) {
   193→    if (n.parentId === ancestorId) return true;
   194→    n = fsNodes.get(n.parentId);
   195→  }
   196→  return false;
   197→}

   365→function createTab(nodeId) {
   366→  const tabId = 't' + (state.nextTabId++);
   367→  const tab = {
   368→    id: tabId,
   369→    nodeId: nodeId,
   370→    history: [nodeId],
   371→    historyIdx: 0,
   372→    selectedIds: new Set(),
   373→  };
   374→  state.tabs.push(tab);
   375→  state.activeTabId = tabId;
   376→  return tab;
   377→}
   378→
   379→function getActiveTab() {
   380→  return state.tabs.find(t => t.id === state.activeTabId);
   381→}
   382→
   383→function switchTab(tabId) {
   384→  state.activeTabId = tabId;
   385→  const tab = getActiveTab();
   386→  if (tab) {
   387→    state.selection = tab.selectedIds;
   388→    state.searchQuery = '';
   389→    document.getElementById('searchInput').value = '';
   390→    renderAll();
   391→  }
   392→}
   393→
   394→function closeTab(tabId) {
   395→  const idx = state.tabs.findIndex(t => t.id === tabId);
   396→  if (idx < 0) return;
   397→  state.tabs.splice(idx, 1);
   398→  if (state.tabs.length === 0) {
   399→    createTab(state.dashboardId);
   400→  } else if (state.activeTabId === tabId) {
   401→    const newTab = state.tabs[Math.min(idx, state.tabs.length - 1)];
   402→    state.activeTabId = newTab.id;
   403→    state.selection = newTab.selectedIds;
   404→  }
   405→  renderAll();
   406→}
   407→
   408→function navigateTo(nodeId) {
   409→  const tab = getActiveTab();
   410→  if (!tab || tab.nodeId === nodeId) return;
   411→  tab.history = tab.history.slice(0, tab.historyIdx + 1);
   412→  tab.history.push(nodeId);
   413→  tab.historyIdx = tab.history.length - 1;
   414→  tab.nodeId = nodeId;
   415→  tab.selectedIds = new Set();
   416→  state.selection = tab.selectedIds;
   417→  state.searchQuery = '';
   418→  document.getElementById('searchInput').value = '';
   419→  renderAll();
   420→}
   421→
   422→function goBack() {
   423→  const tab = getActiveTab();
   424→  if (!tab || tab.historyIdx <= 0) return;
   425→  tab.historyIdx--;
   426→  tab.nodeId = tab.history[tab.historyIdx];
   427→  tab.selectedIds = new Set();
   428→  state.selection = tab.selectedIds;
   429→  renderAll();
   430→}
   431→
   432→function goForward() {
   433→  const tab = getActiveTab();
   434→  if (!tab || tab.historyIdx >= tab.history.length - 1) return;
   435→  tab.historyIdx++;
   436→  tab.nodeId = tab.history[tab.historyIdx];
   437→  tab.selectedIds = new Set();
   438→  state.selection = tab.selectedIds;
   439→  renderAll();
   440→}
   441→
   442→function goUp() {
   443→  const tab = getActiveTab();
   444→  if (!tab) return;
   445→  const node = fsNodes.get(tab.nodeId);
   446→  if (node && node.parentId) {

   568→function renderTreeNode(nodeId, depth, activeNodeId) {
   569→  const node = fsNodes.get(nodeId);
   570→  if (!node) return '';
   571→
   572→  const hasChildren = node.children && node.children.length > 0 && node.children.some(cid => {
   573→    const c = fsNodes.get(cid);
   574→    return c && c.type === 'folder';
   575→  });
   576→  const isActive = nodeId === activeNodeId;
   577→  const padLeft = 8 + depth * 18;
   578→
   579→  let html = `<div class="tree-node">`;
   580→  html += `<div class="tree-item ${isActive ? 'active' : ''}" data-node-id="${nodeId}" style="padding-left:${padLeft}px">`;

   582→    <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><polyline points="9 18 15 12 9 6"></polyline></svg>
   583→  </span>`;
   584→  html += `<i data-lucide="${node.expanded ? 'folder-open' : 'folder'}" class="tree-icon"></i>`;
   585→  html += `<span class="tree-label">${escapeHtml(node.name)}</span>`;
   586→  html += `</div>`;
   587→
   588→  if (hasChildren && node.expanded) {
   589→    html += `<div class="tree-children expanded">`;
   590→    for (const cid of node.children) {
   591→      const child = fsNodes.get(cid);
   592→      if (child && child.type === 'folder') {
   593→        html += renderTreeNode(cid, depth + 1, activeNodeId);
   594→      }
   595→    }
   596→    html += `</div>`;
   597→  }
   598→
   599→  html += `</div>`;
   600→  return html;

   859→function getCurrentItems() {
   860→  const tab = getActiveTab();
   861→  if (!tab || tab.nodeId === 'dashboard') return [];
   862→  
   863→  const node = fsNodes.get(tab.nodeId);
   864→  if (!node || !node.children) return [];
   865→  let items = node.children.map(cid => fsNodes.get(cid)).filter(Boolean);
   866→
   867→  if (state.searchQuery) {
   868→    const q = state.searchQuery.toLowerCase();
   869→    items = items.filter(it => it.name.toLowerCase().includes(q));
   870→  }
   871→
   872→  const dir = state.sortDir === 'asc' ? 1 : -1;
   873→  items.sort((a, b) => {
   874→    const aFolder = a.type === 'folder';
   875→    const bFolder = b.type === 'folder';
   876→    if (aFolder !== bFolder) return aFolder ? -1 : 1;
   877→    let av, bv;
   878→    switch (state.sortBy) {
   879→      case 'name':
   880→        av = a.name.toLowerCase(); bv = b.name.toLowerCase();
   881→        return av < bv ? -1 * dir : (av > bv ? 1 * dir : 0);
   882→      case 'size':
   883→        av = a.size; bv = b.size;
   884→        return (av - bv) * dir;
   885→      case 'type':
   886→        av = a.type; bv = b.type;
   887→        return av < bv ? -1 * dir : (av > bv ? 1 * dir : 0);
   888→      case 'modified':
   889→        av = a.modified.getTime(); bv = b.modified.getTime();
   890→        return (av - bv) * dir;
   891→      default: return 0;
   892→    }
   893→  });
   894→
   895→  return items;
   896→}

   898→function formatSize(bytes) {
   899→  if (bytes === 0) return '—';
   900→  if (bytes < 1024) return bytes + ' B';
   901→  if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB';
   902→  if (bytes < 1024 * 1024 * 1024) return (bytes / 1024 / 1024).toFixed(1) + ' MB';
   903→  if (bytes < 1024 * 1024 * 1024 * 1024) return (bytes / 1024 / 1024 / 1024).toFixed(2) + ' GB';
   904→  return (bytes / 1024 / 1024 / 1024 / 1024).toFixed(2) + ' TB';
   905→}

   907→function formatDate(d) {
   908→  const now = new Date();
   909→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   910→  const that = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   911→  const diffDays = Math.round((today - that) / 86400000);
   912→  if (diffDays === 0) return 'Today, ' + d.toLocaleTimeString([], {hour: '2-digit', minute: '2-digit'});
   913→  if (diffDays === 1) return 'Yesterday';
   914→  if (diffDays < 7) return diffDays + ' days ago';
   915→  return d.toLocaleDateString([], {year: 'numeric', month: 'short', day: 'numeric'});
   916→}

   981→function wireFileItems() {
   982→  document.querySelectorAll('.file-item, .list-row').forEach(el => {
   983→    const nodeId = el.dataset.nodeId;
   984→
   985→    el.addEventListener('click', (e) => {
   986→      const isCheckbox = e.target.closest('.file-checkbox');
   987→
   988→      if (isCheckbox) {
   989→        e.stopPropagation();
   990→        if (state.selection.has(nodeId)) state.selection.delete(nodeId);
   991→        else state.selection.add(nodeId);
   992→      } else if (e.ctrlKey || e.metaKey) {
   993→        if (state.selection.has(nodeId)) state.selection.delete(nodeId);
   994→        else state.selection.add(nodeId);
   995→      } else if (e.shiftKey) {
   996→        const items = getCurrentItems();
   997→        const idx = items.findIndex(i => i.id === nodeId);
   998→        if (idx >= 0 && state.selection.size > 0) {
   999→          const lastSelected = Array.from(state.selection).pop();
  1000→          const lastIdx = items.findIndex(i => i.id === lastSelected);
  1001→          if (lastIdx >= 0) {
  1002→            const start = Math.min(idx, lastIdx);
  1003→            const end = Math.max(idx, lastIdx);
  1004→            for (let i = start; i <= end; i++) state.selection.add(items[i].id);
  1005→          }
  1006→        } else {
  1007→          state.selection.clear();
  1008→          state.selection.add(nodeId);
  1009→        }
  1010→      } else {
  1011→        state.selection.clear();
  1012→        state.selection.add(nodeId);
  1013→      }
  1014→
  1015→      const tab = getActiveTab();
  1016→      if (tab) tab.selectedIds = state.selection;
  1017→      updateSelectionUI();
  1018→    });
  1019→
  1020→    el.addEventListener('dblclick', () => {
  1021→      const node = fsNodes.get(nodeId);
  1022→      if (!node) return;
  1023→      if (node.type === 'folder') {
  1024→        navigateTo(nodeId);
  1025→      } else {
  1026→        showProperties(nodeId);
  1027→      }
  1028→    });
  1029→
  1030→    el.addEventListener('contextmenu', (e) => {
  1031→      e.preventDefault();
  1032→      if (!state.selection.has(nodeId)) {
  1033→        state.selection.clear();
  1034→        state.selection.add(nodeId);
  1035→        const tab = getActiveTab();
  1036→        if (tab) tab.selectedIds = state.selection;
  1037→        updateSelectionUI();
  1038→      }
  1039→      showFileContextMenu(e.clientX, e.clientY, nodeId);
  1040→    });
  1041→  });
  1042→
  1043→  document.querySelectorAll('.list-header .col').forEach(c => {
  1044→    c.addEventListener('click', () => {
  1045→      const sort = c.dataset.sort;
  1046→      if (!sort) return;
  1047→      if (state.sortBy === sort) {
  1048→        state.sortDir = state.sortDir === 'asc' ? 'desc' : 'asc';
  1049→      } else {
  1050→        state.sortBy = sort;
  1051→        state.sortDir = 'asc';
  1052→      }

  1054→      renderStatusbar();
  1055→    });
  1056→  });

  1587→function pasteClipboard() {
  1588→  if (!state.clipboard) return;
  1589→  const tab = getActiveTab();
  1590→  if (!tab || tab.nodeId === 'dashboard') return;
  1591→  
  1592→  const dest = fsNodes.get(tab.nodeId);
  1593→  if (!dest || !dest.children) return;
  1594→
  1595→  let count = 0;
  1596→  for (const id of state.clipboard.ids) {
  1597→    const node = fsNodes.get(id);
  1598→    if (!node) continue;
  1599→    if (state.clipboard.mode === 'cut' && (id === tab.nodeId || isAncestor(id, tab.nodeId))) continue;
  1600→
  1601→    let newName = node.name;
  1602→    if (state.clipboard.mode === 'copy') {
  1603→      newName = getUniqueName(dest, node.name);
  1604→      cloneNodeTree(node, dest, newName);
  1605→    } else {
  1606→      const oldParent = fsNodes.get(node.parentId);
  1607→      if (oldParent) {
  1608→        const idx = oldParent.children.indexOf(id);
  1609→        if (idx >= 0) oldParent.children.splice(idx, 1);
  1610→      }
  1611→      node.parentId = dest.id;
  1612→      node.name = newName;
  1613→      dest.children.push(id);
  1614→    }
  1615→    count++;
  1616→  }
  1617→
  1618→  if (state.clipboard.mode === 'cut') state.clipboard = null;
  1619→  state.selection.clear();
  1620→  renderAll();
  1621→  if (count > 0) showToast(`Pasted ${count} item${count > 1 ? 's' : ''}`, 'success');
  1622→}

  1624→function cloneNodeTree(node, parent, name) {
  1625→  const clone = createNode({
  1626→    name: name,
  1627→    type: node.type,
  1628→    parentId: parent.id,
  1629→    size: node.size,
  1630→    meta: { ...node.meta },
  1631→  });
  1632→  parent.children.push(clone.id);
  1633→  if (node.children) {
  1634→    for (const cid of node.children) {
  1635→      const child = fsNodes.get(cid);
  1636→      if (child) cloneNodeTree(child, clone, child.name);
  1637→    }

  1642→function getUniqueName(parent, name) {
  1643→  if (!getChildByName(parent.id, name)) return name;
  1644→  const dot = name.lastIndexOf('.');
  1645→  const base = dot > 0 ? name.substring(0, dot) : name;
  1646→  const ext = dot > 0 ? name.substring(dot) : '';
  1647→  let i = 1;
  1648→  while (getChildByName(parent.id, `${base} (copy${i > 1 ? ' ' + i : ''})${ext}`)) i++;
  1649→  return `${base} (copy${i > 1 ? ' ' + i : ''})${ext}`;
  1650→}
