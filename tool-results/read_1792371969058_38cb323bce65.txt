   623→  1523→function createNode(data) {
   624→  1524→  const id = data.id || ('n' + (nextId++));
   625→  1525→  const node = {
   626→  1526→    id,
   627→  1527→    name: data.name,
   628→  1528→    type: data.type || 'folder',
   629→  1529→    parentId: data.parentId || null,
   630→  1530→    children: (data.type === undefined || data.type === 'folder') ? (data.children || []) : null,
   631→  1531→    size: data.size || 0,
   632→  1532→    modified: data.modified || randomDate(),
   633→  1533→    meta: data.meta || {},
   634→  1534→    expanded: data.expanded || false,
   635→  1535→  };
   636→  1536→  fsNodes.set(id, node);
   637→  1537→  return node;
   638→  1538→}

   640→  1540→function randomDate() {
   641→  1541→  const days = Math.floor(Math.random() * 120);
   642→  1542→  const d = new Date(Date.now() - days * 86400000 - Math.random() * 86400000);
   643→  1543→  return d;
   644→  1544→}

   646→  1546→function getChildByName(parentId, name) {
   647→  1547→  const parent = fsNodes.get(parentId);
   648→  1548→  if (!parent || !parent.children) return null;
   649→  1549→  for (const cid of parent.children) {
   650→  1550→    const c = fsNodes.get(cid);
   651→  1551→    if (c && c.name === name) return c;
   652→  1552→  }
   653→  1553→  return null;
   654→  1554→}
   655→  1555→
   656→  1556→function getPath(nodeId) {
   657→  1557→  const parts = [];
   658→  1558→  let n = fsNodes.get(nodeId);
   659→  1559→  while (n) {
   660→  1560→    parts.unshift({ id: n.id, name: n.name });
   661→  1561→    n = n.parentId ? fsNodes.get(n.parentId) : null;
   662→  1562→  }
   663→  1563→  return parts;
   664→  1564→}
   665→  1565→
   666→  1566→function findNodeByPath(parts) {
   667→  1567→  let cur = fsNodes.get('root');
   668→  1568→  for (const p of parts) {
   669→  1569→    if (!cur || !cur.children) return null;
   670→  1570→    let found = null;
   671→  1571→    for (const cid of cur.children) {
   672→  1572→      const c = fsNodes.get(cid);
   673→  1573→      if (c && c.name === p) { found = c; break; }
   674→  1574→    }
   675→  1575→    if (!found) return null;
   676→  1576→    cur = found;
   677→  1577→  }
   678→  1578→  return cur ? cur.id : null;
   679→  1579→}

   681→  1581→function isAncestor(ancestorId, nodeId) {
   682→  1582→  let n = fsNodes.get(nodeId);
   683→  1583→  while (n && n.parentId) {
   684→  1584→    if (n.parentId === ancestorId) return true;
   685→  1585→    n = fsNodes.get(n.parentId);
   686→  1586→  }
   687→  1587→  return false;
   688→  1588→}
