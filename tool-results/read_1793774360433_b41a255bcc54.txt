     2→ * de-shell.js — <de-shell>
     3→ *
     4→ * The orchestrating host. Shadow tree contains: wallpaper, <de-music-widget>,

     7→ * This is the ONLY component that owns the design tokens (the original

    18→ * For each open window the shell creates a <de-app-window data-window-id=…
    19→ * title=… icon=…> containing <fe-app> (appId==='files') or

   322→function hexToRgb(hex) {
   323→  const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
   324→  return result ? {
   325→    r: parseInt(result[1], 16),
   326→    g: parseInt(result[2], 16),
   327→    b: parseInt(result[3], 16)
   328→  } : null;

   331→class DeShell extends HTMLElement {
   332→  connectedCallback() {
   333→    if (this.shadowRoot) return;
   334→    const root = this.attachShadow({ mode: 'open' });
   335→    root.innerHTML = CSS + HTML;
   336→    this._root = root;
   337→    requestAnimationFrame(() => this._init());
   338→  }
   339→
   340→  _init() {
   341→    this._initIcons = () => renderIcons(this._root);
   342→    this._initIcons();
   343→
   344→    // child refs
   345→    this._dock = this._root.querySelector('de-dock');
   346→    this._launcher = this._root.querySelector('de-launcher');
   347→    this._qs = this._root.querySelector('de-quicksettings');
   348→    this._cal = this._root.querySelector('de-calendar');
   349→    this._windowsLayer = this._root.getElementById('windows-layer');
   350→    this._desktopWidgets = this._root.getElementById('desktop-widgets');
   351→    this._gridOverlay = this._root.getElementById('grid-overlay');
   352→    this._snapGuides = this._root.getElementById('snap-guides');
   353→    this._selectionRect = this._root.getElementById('selection-rect');
   354→    this._contextMenu = this._root.getElementById('widget-context-menu');
   355→    this._selectedWidgets = new Set(); // selected desktop widget ids
   356→    this._widgetZTop = 10; // z-order counter for desktop widgets
   357→
   358→    // desktop-form transformer widgets (floating panels)
   359→    this._renderDesktopWidgets();
   360→    this._unsubDesktopWidgets = onWidgets(() => this._renderDesktopWidgets());
   361→    // desktop is a drop target for widgets (drag from the bento grid or the bar)
   362→    this._wireDesktopDrop();
   363→    // desktop selection rectangle (drag on empty desktop) + context menu close
   364→    this._wireDesktopSelection();
   365→
   366→    // subscribe to window-service
   367→    this._unsubWindows = onWindowsChange((state) => this._renderWindows(state));
   368→    this._renderWindows(windowService.state);
   369→
   370→    // re-render windows on viewport resize so tiling tiles recompute
   371→    this._resizeHandler = () => this._renderWindows(windowService.state);
   372→    window.addEventListener('resize', this._resizeHandler);
   373→
   374→    // subscribe to theme-service
   375→    this._unsubTheme = onTheme((state) => this._applyTheme(state));
   376→    this._applyTheme(themeService.state);
   377→
   378→    // subscribe to ui.popover bus
   379→    this._unsubBus = bus.on('ui.popover', (payload) => this._handlePopover(payload));
   380→
   381→    // escape closes any open popover
   382→    this._keyHandler = (e) => {
   383→      if (e.key === 'Escape') {
   384→        if (this._anyOpen()) commands.execute('ui.closeAllPopovers');
   385→      }
   386→    };
   387→    document.addEventListener('keydown', this._keyHandler);
   388→
   389→    // late icon refresh (CDN may load late)
   390→    setTimeout(() => this._initIcons(), 50);
   391→  }
   392→
   393→  _anyOpen() {
   394→    return (
   395→      (this._launcher && this._launcher.isOpen()) ||
   396→      (this._qs && this._qs.isOpen()) ||
   397→      (this._cal && this._cal.isOpen()) ||
   398→      (this._dock && this._dock.isThemePopupOpen()) ||
   399→      (this._dock && this._dock.isWidgetPopoverOpen())
   400→    );
   401→  }
   402→
   403→  _renderWindows(state) {
   404→    const layer = this._windowsLayer;
   405→    if (!layer) return;
   406→    // track only <de-app-window> elements (NOT tile-dividers) by windowId
   407→    const existing = new Map();
   408→    Array.from(layer.children).forEach((el) => {
   409→      if (el.dataset.windowId) existing.set(el.dataset.windowId, el);
   410→    });
   411→
   412→    const area = this._tilingArea();
   413→    const activeWs = workspaceService.activeId;
   414→    const layout = state.layoutMode;
   415→
   416→    // tiles are computed only for the ACTIVE workspace's non-minimized windows
   417→    const activeWindows = state.windows.filter((w) => !w.minimized);
   418→    const tiling = (layout === 'horizontal' || layout === 'vertical');
   419→    const tiles = tiling ? this._computeTiles(activeWindows, area, layout) : null;
   420→    let tileIdx = 0;
   421→
   422→    // iterate ALL windows (across every workspace) so elements persist on switch
   423→    for (const w of windowService.allWindows) {
   424→      let el = existing.get(w.id);
   425→      if (!el) {
   426→        el = document.createElement('de-app-window');
   427→        el.dataset.windowId = w.id;
   428→        el.dataset.appId = w.appId;
   429→        if (w.titlebar === false) el.setAttribute('data-notitlebar', '');
   430→        el.setAttribute('title', w.title || w.appId);
   431→        if (w.icon) el.setAttribute('icon', w.icon);
   432→        if (w.appId === 'files') {
   433→          el.appendChild(document.createElement('fe-app'));
   434→        } else {
   435→          const ph = document.createElement('de-placeholder-app');
   436→          ph.dataset.appId = w.appId;
   437→          el.appendChild(ph);
   438→        }
   439→        layer.appendChild(el);
   440→      } else {
   441→        existing.delete(w.id);
   442→        el.setAttribute('title', w.title || w.appId);
   443→        if (w.icon) el.setAttribute('icon', w.icon);
   444→      }
   445→
   446→      const inActiveWs = (w.workspaceId === activeWs);
   447→      // hide windows that belong to other workspaces (element is kept alive)
   448→      if (!inActiveWs) {
   449→        el.style.display = 'none';
   450→        continue;
   451→      }
   452→
   453→      el.style.zIndex = String(w.z);
   454→      el.classList.toggle('focused', state.focusedId === w.id);
   455→      el.setAttribute('data-layout', layout);
   456→      if (w.maximized) el.setAttribute('data-maximized', '');
   457→      else el.removeAttribute('data-maximized');
   458→      el.style.display = w.minimized ? 'none' : '';
   459→
   460→      let r;
   461→      if (w.maximized) {
   462→        r = { x: area.x, y: area.y, w: area.w, h: area.h };
   463→      } else if (tiling) {
   464→        if (w.minimized) { r = null; }
   465→        else { r = tiles[tileIdx] || { x: area.x, y: area.y, w: area.w, h: area.h }; tileIdx++; }
   466→      } else {
   467→        r = { x: w.x ?? 80, y: w.y ?? 24, w: w.w ?? 1000, h: w.h ?? 620 };
   468→      }
   469→      if (r) {
   470→        el.style.setProperty('--win-x', r.x + 'px');
   471→        el.style.setProperty('--win-y', r.y + 'px');
   472→        el.style.setProperty('--win-w', r.w + 'px');
   473→        el.style.setProperty('--win-h', r.h + 'px');
   474→      }
   475→    }
   476→
   477→    // remove <de-app-window> elements whose windows no longer exist anywhere
   478→    for (const [, el] of existing) el.remove();
   479→
   480→    // draggable dividers between adjacent tiles (horizontal / vertical only)
   481→    this._renderTilingDividers(tiling ? tiles : [], activeWindows, layout, area);
   482→
   483→    // mark active app on dock (active workspace only)
   484→    if (this._dock) {
   485→      const focused = state.windows.find((w) => w.id === state.focusedId);
   486→      if (focused) this._dock.setActiveApp(focused.appId);
   487→      else this._dock.clearActiveApp();
   488→    }
   489→  }

   491→  /** the area above the dock where tiled windows live */
   492→  _tilingArea() {
   493→    const vw = window.innerWidth, vh = window.innerHeight;
   494→    return { x: 8, y: 8, w: vw - 16, h: vh - 8 - 84 }; // 84 = dock + gap
   495→  }

   497→  /**
   498→   * Tiling tiles using each window's tileRatio (so dragging a divider resizes
   499→   * adjacent tiles). 'horizontal' = one row of N columns; 'vertical' = one
   500→   * column of N rows; small gaps between tiles.
   501→   */
   502→  _computeTiles(windows, area, layout) {
   503→    if (!windows.length) return [];
   504→    const gap = 6;
   505→    const ratios = windows.map((w) => w.tileRatio || 1);
   506→    const sumR = ratios.reduce((a, b) => a + b, 0) || 1;
   507→    const tiles = [];
   508→    if (layout === 'horizontal') {
   509→      const usableW = area.w - (windows.length - 1) * gap;
   510→      let cursor = area.x;
   511→      windows.forEach((w, i) => {
   512→        const tw = usableW * (ratios[i] / sumR);
   513→        tiles.push({ x: cursor, y: area.y, w: tw, h: area.h });
   514→        cursor += tw + gap;
   515→      });
   516→    } else { // vertical
   517→      const usableH = area.h - (windows.length - 1) * gap;
   518→      let cursor = area.y;
   519→      windows.forEach((w, i) => {
   520→        const th = usableH * (ratios[i] / sumR);
   521→        tiles.push({ x: area.x, y: cursor, w: area.w, h: th });
   522→        cursor += th + gap;
   523→      });
   524→    }
   525→    return tiles;
   526→  }

   528→  /**
   529→   * Render / update draggable dividers between adjacent tiles. Dividers are
   530→   * diffed by data-key (a|b) so they persist across re-renders (e.g. while
   531→   * dragging one, the ratios change and tiles recompute, but the divider
   532→   * element stays and just repositions).
   533→   */
   534→  _renderTilingDividers(tiles, activeWindows, layout, area) {
   535→    const layer = this._windowsLayer;
   536→    if (!layer) return;
   537→    const existing = new Map();
   538→    layer.querySelectorAll('.tile-divider').forEach((d) => existing.set(d.dataset.key, d));
   539→    const wanted = new Set();
   540→
   541→    for (let i = 0; i < activeWindows.length - 1; i++) {
   542→      const a = activeWindows[i], b = activeWindows[i + 1];
   543→      const key = a.id + '|' + b.id;
   544→      wanted.add(key);
   545→      let div = existing.get(key);
   546→      if (!div) {
   547→        div = document.createElement('div');
   548→        div.dataset.key = key;
   549→        div.dataset.a = a.id;
   550→        div.dataset.b = b.id;
   551→        this._wireDivider(div);
   552→        layer.appendChild(div);
   553→      }
   554→      div.className = 'tile-divider ' + (layout === 'horizontal' ? 'h' : 'v');
   555→      const ta = tiles[i];
   556→      const gap = 6;
   557→      if (layout === 'horizontal') {
   558→        div.style.left = (ta.x + ta.w + gap / 2 - 4) + 'px';
   559→        div.style.top = ta.y + 'px';
   560→        div.style.width = '8px';
   561→        div.style.height = ta.h + 'px';
   562→      } else {
   563→        div.style.top = (ta.y + ta.h + gap / 2 - 4) + 'px';
   564→        div.style.left = ta.x + 'px';
   565→        div.style.width = ta.w + 'px';
   566→        div.style.height = '8px';
   567→      }
   568→    }
   569→    for (const [key, div] of existing) if (!wanted.has(key)) div.remove();
   570→  }

   572→  /** wire a tile-divider drag → adjust the two adjacent windows' tileRatios */
   573→  _wireDivider(div) {
   574→    div.style.position = 'absolute';
   575→    div.style.zIndex = '100';
   576→    div.addEventListener('pointerdown', (e) => {
   577→      e.stopPropagation();
   578→      e.preventDefault();
   579→      const isH = div.classList.contains('h');
   580→      const aId = div.dataset.a, bId = div.dataset.b;
   581→      const ws = workspaceService.active;
   582→      const wa = ws.windows.find((w) => w.id === aId);
   583→      const wb = ws.windows.find((w) => w.id === bId);
   584→      if (!wa || !wb) return;
   585→      const ra0 = wa.tileRatio || 1, rb0 = wb.tileRatio || 1;
   586→      const sumR = ra0 + rb0;
   587→      const area = this._tilingArea();
   588→      const nVis = ws.windows.filter((w) => !w.minimized).length;
   589→      const usable = (isH ? area.w : area.h) - (nVis - 1) * 6;
   590→      const startX = e.clientX, startY = e.clientY;
   591→      const onMove = (ev) => {
   592→        const d = isH ? (ev.clientX - startX) : (ev.clientY - startY);
   593→        const delta = (d / usable) * sumR;
   594→        let na = ra0 + delta, nb = rb0 - delta;
   595→        const minR = 0.15 * sumR;
   596→        if (na < minR) { na = minR; nb = sumR - minR; }
   597→        if (nb < minR) { nb = minR; na = sumR - minR; }
   598→        commands.execute('window.setTileRatios', { list: [{ id: aId, ratio: na }, { id: bId, ratio: nb }] });
   599→      };
   600→      const onUp = () => {
   601→        document.removeEventListener('pointermove', onMove);
   602→        document.removeEventListener('pointerup', onUp);
   603→      };
   604→      document.addEventListener('pointermove', onMove);
   605→      document.addEventListener('pointerup', onUp);
   606→    });
   607→  }

   662→  /** wire a desktop panel's body drag with Ctrl grid-snap + Shift widget-snap +
   663→   *  multi-selection (Ctrl+click) + context menu (right-click). */
   664→  _wirePanelMove(panel, id) {
   665→    panel.addEventListener('pointerdown', (e) => {
   666→      if (e.target.closest('.rz-handle, .dw-close, button, input, a')) return;
   667→      if (e.button === 2) return; // right-click handled by contextmenu
   668→      e.preventDefault();
   669→      const sx = e.clientX, sy = e.clientY;
   670→      const startLeft = panel.offsetLeft, startTop = panel.offsetTop;
   671→      const vw = window.innerWidth, vh = window.innerHeight;
   672→      const dockEl = this._root.querySelector('de-dock');
   673→      const dockContainer = dockEl && dockEl.shadowRoot && dockEl.shadowRoot.getElementById('dock-container');
   674→      const overBar = (y) => { if (!dockContainer) return false; const r = dockContainer.getBoundingClientRect(); return y >= r.top - 10 && y <= r.bottom + 10; };
   675→      let mode = 'idle';
   676→      const ctrlAtDown = e.ctrlKey || e.metaKey;
   677→
   678→      const move = (ev) => {
   679→        if (mode === 'idle') {
   680→          if (Math.hypot(ev.clientX - sx, ev.clientY - sy) < 4) return;
   681→          if (overBar(ev.clientY)) { mode = 'widget'; panel.classList.add('dragging'); startWidgetDrag(id, 'desktop'); return; }
   682→          mode = 'move';
   683→          if (!ctrlAtDown) this._clearSelection();
   684→          this._selectOnly(panel, id);
   685→        }
   686→        if (mode !== 'move') return;
   687→        let nx = startLeft + (ev.clientX - sx);
   688→        let ny = startTop + (ev.clientY - sy);
   689→        // Ctrl → grid snap + show dot grid
   690→        if (ev.ctrlKey || ev.metaKey) {
   691→          this._showGrid();
   692→          nx = snapToGrid(nx); ny = snapToGrid(ny);
   693→        } else this._hideGrid();
   694→        // Shift → snap to other widgets + show guides
   695→        if (ev.shiftKey) {
   696→          const others = this._getOtherPanels(panel);
   697→          const g = findSnapGuides({ left: nx, top: ny, w: panel.offsetWidth, h: panel.offsetHeight }, others);
   698→          nx += g.dx; ny += g.dy;
   699→          this._showGuides(g);
   700→        } else this._hideGuides();
   701→        nx = Math.max(4, Math.min(vw - panel.offsetWidth - 4, nx));
   702→        ny = Math.max(4, Math.min(vh - 100, ny));
   703→        panel.style.left = nx + 'px';
   704→        panel.style.top = ny + 'px';
   705→      };
   706→      const up = () => {
   707→        document.removeEventListener('pointermove', move);
   708→        document.removeEventListener('pointerup', up);
   709→        panel.classList.remove('dragging');
   710→        this._hideGrid(); this._hideGuides();
   711→        if (mode === 'move') widgetService.setDesktopSize(id, panel.offsetWidth, panel.offsetHeight, panel.offsetLeft, panel.offsetTop);
   712→        if (mode === 'idle') { // click without drag
   713→          if (ctrlAtDown) this._toggleSelection(panel, id);
   714→          else { this._clearSelection(); this._selectOnly(panel, id); }
   715→        }
   716→      };
   717→      document.addEventListener('pointermove', move);
   718→      document.addEventListener('pointerup', up);
   719→    });
   720→    // right-click → context menu (align/distribute)
   721→    panel.addEventListener('contextmenu', (e) => {
   722→      e.preventDefault();
   723→      if (!panel.classList.contains('selected')) { this._clearSelection(); this._selectOnly(panel, id); }
   724→      this._showContextMenu(e.clientX, e.clientY);
   725→    });
   726→  }

   740→  /** snap a resize to the grid (Ctrl) or other widgets (Shift) */
   741→  _snapResize({ x, y, w, h }, ev, dir, panel) {
   742→    if (ev.ctrlKey || ev.metaKey) {
   743→      this._showGrid();
   744→      w = snapToGrid(w); h = snapToGrid(h);
   745→      if (dir.includes('w')) x = snapToGrid(x);
   746→      if (dir.includes('n')) y = snapToGrid(y);
   747→    } else this._hideGrid();
   748→    if (ev.shiftKey) {
   749→      const others = this._getOtherPanels(panel);
   750→      const g = findSnapGuides({ left: x, top: y, w, h }, others);
   751→      x += g.dx; y += g.dy;
   752→      if (dir.includes('w')) w -= g.dx;
   753→      if (dir.includes('n')) h -= g.dy;
   754→      this._showGuides(g);
   755→    } else this._hideGuides();
   756→    return { x, y, w: Math.max(200, w), h: Math.max(100, h) };
   757→  }

   848→  /** drag-select rectangle on empty desktop */

   850→    const desktop = this._root.querySelector('.desktop');
   851→    const rect = this._selectionRect;
   852→    if (!desktop || !rect) return;
   853→    desktop.addEventListener('pointerdown', (e) => {
   854→      // only start selection on truly empty desktop (not on panels, dock, popovers)
   855→      if (e.target.closest('.dw-panel, de-dock, de-launcher, de-quicksettings, de-calendar, #widget-context-menu, .tile-divider, #widget-popover')) return;
   856→      const sx = e.clientX, sy = e.clientY;
   857→      let dragging = false;
   858→      const move = (ev) => {
   859→        if (!dragging && Math.hypot(ev.clientX - sx, ev.clientY - sy) < 4) return;
   860→        dragging = true;
   861→        const x = Math.min(sx, ev.clientX), y = Math.min(sy, ev.clientY);
   862→        const w = Math.abs(ev.clientX - sx), h = Math.abs(ev.clientY - sy);
   863→        rect.style.left = x + 'px'; rect.style.top = y + 'px'; rect.style.width = w + 'px'; rect.style.height = h + 'px';
   864→        rect.classList.add('visible');
   865→      };
   866→      const up = (ev) => {
   867→        document.removeEventListener('pointermove', move);
   868→        document.removeEventListener('pointerup', up);
   869→        rect.classList.remove('visible'); rect.style.width = '0'; rect.style.height = '0';
   870→        if (!dragging) { this._clearSelection(); return; }
   871→        const rx = Math.min(sx, ev.clientX), ry = Math.min(sy, ev.clientY);
   872→        const rw = Math.abs(ev.clientX - sx), rh = Math.abs(ev.clientY - sy);
   873→        this._clearSelection();
   874→        this._desktopWidgets.querySelectorAll('.dw-panel').forEach((p) => {
   875→          const px = p.offsetLeft, py = p.offsetTop, pw = p.offsetWidth, ph = p.offsetHeight;
   876→          if (px < rx + rw && px + pw > rx && py < ry + rh && py + ph > ry) {
   877→            p.classList.add('selected');
   878→            this._selectedWidgets.add(p.querySelector('tf-widget')?.dataset.id);
   879→          }
   880→        });
   881→      };
   882→      document.addEventListener('pointermove', move);
   883→      document.addEventListener('pointerup', up);
   884→    });
   885→    // close context menu on outside click
   886→    document.addEventListener('click', (e) => {
   887→      if (this._contextMenu?.classList.contains('visible') && !e.composedPath().includes(this._contextMenu)) this._hideContextMenu();
   888→    });
   889→  }
   890→
   891→  /** make the desktop background a drop target for the pointer-based widget drag
   892→   *  (bus-coordinated). Shows a subtle highlight while dragging over it. */
   893→  _wireDesktopDrop() {
   894→    const desktop = this._root.querySelector('.desktop');
   895→    if (!desktop) return;
   896→    // the drop area = the desktop excluding the dock strip (bottom 84px) so a
   897→    // drop near the dock doesn't accidentally target the desktop.
   898→    const overDesktop = (x, y) => {
   899→      const r = desktop.getBoundingClientRect();
   900→      return x >= r.left && x <= r.right && y >= r.top && y <= r.bottom - 84;
   901→    };
   902→    this._unsubDeskDragMove = bus.on('widget.dragmove', ({ x, y }) => {
   903→      desktop.classList.toggle('desktop-drop-hover', overDesktop(x, y));
   904→    });
   905→    this._unsubDeskDragEnd = bus.on('widget.dragend', ({ x, y, id, from }) => {
   906→      desktop.classList.remove('desktop-drop-hover');
   907→      if (overDesktop(x, y)) {
   908→        if (from === 'palette') widgetService.addToContainer(id, 'desktop');
   909→        else if (from === 'bar') widgetService.moveWidget(id, 'bar', 'desktop');
   910→        else if (from === 'desktop') {
   911→          // reposition the desktop panel to the drop point (drag-to-move via the
   912→          // widget-drag ghost — same UX as dragging to the bar)
   913→          const tw = this._desktopWidgets.querySelector(`tf-widget[data-id="${id}"]`);
   914→          const panel = tw && tw.closest('.dw-panel');
   915→          if (panel) {
   916→            const w = panel.offsetWidth, h = panel.offsetHeight;
   917→            const nx = Math.max(4, Math.min(window.innerWidth - w - 4, x - w / 2));
   918→            const ny = Math.max(4, Math.min(window.innerHeight - 100, y - h / 2));
   919→            panel.style.left = nx + 'px';
   920→            panel.style.top = ny + 'px';
   921→            widgetService.setDesktopSize(id, w, h, nx, ny);
   922→          }
   923→        }
   924→      }
   925→    });
   926→  }
   927→
   928→  _applyTheme(state) {
   929→    const theme = THEMES[state.theme];
   930→    if (!theme) return;
   931→    const isLight = state.mode === 'light';
   932→    // toggle light class on host
   933→    this.classList.toggle('light', isLight);
   934→    // write --accent-* on this.style so children inherit
   935→    const accent = theme.accent;
   936→    this.style.setProperty('--accent-strong', accent);
   937→    const rgb = hexToRgb(accent);
   938→    if (rgb) {
   939→      this.style.setProperty('--accent', `rgba(${rgb.r},${rgb.g},${rgb.b},0.8)`);
   940→      this.style.setProperty('--accent-light', `rgba(${rgb.r},${rgb.g},${rgb.b},1)`);
   941→      this.style.setProperty('--accent-glow', `rgba(${rgb.r},${rgb.g},${rgb.b},0.35)`);
   942→    }
   943→    // wallpaper
   944→    this._updateWallpaper(theme, isLight);
   945→  }
   946→
   947→  _updateWallpaper(theme, isLight) {
   948→    const url = isLight ? theme.light.url : theme.dark.url;
   949→    const img = this._root.getElementById('wallpaperImg');
   950→    if (!img) return;
   951→    if (url) {
   952→      const temp = new Image();
   953→      temp.onload = () => {
   954→        img.src = url;
   955→        img.style.opacity = '0.85';
   956→      };
   957→      temp.src = url;
   958→    } else {
   959→      img.style.opacity = '0';
   960→    }
   961→  }

   963→  /** close every popover EXCEPT the one named `which` (null = close all) */
   964→  _closeAllExcept(which) {
   965→    if (which !== 'launcher' && this._launcher) this._launcher.close();
   966→    if (which !== 'qs' && this._qs) this._qs.close();
   967→    if (which !== 'cal' && this._cal) this._cal.close();
   968→    if (which !== 'theme' && this._dock) this._dock.closeThemePopup();
   969→    if (which !== 'widget' && this._dock) this._dock._closeWidgetPopover();
   970→  }
   971→
   972→  _handlePopover({ which, action, anchor } = {}) {
   973→    if (action === 'closeAll') { this._closeAllExcept(null); return; }
   974→    if (action === 'toggle') {
   975→      const wasOpen = this._isOpen(which);
   976→      this._closeAllExcept(which);
   977→      if (!wasOpen) this._open(which, anchor);
   978→    } else if (action === 'open') {
   979→      this._closeAllExcept(which);
   980→      this._open(which, anchor);
   981→    } else if (action === 'close') {
   982→      this._close(which);
   983→    }
   984→  }
   985→
   986→  _isOpen(which) {
   987→    if (which === 'launcher') return this._launcher && this._launcher.isOpen();
   988→    if (which === 'qs') return this._qs && this._qs.isOpen();
   989→    if (which === 'cal') return this._cal && this._cal.isOpen();
   990→    if (which === 'theme') return this._dock && this._dock.isThemePopupOpen();
   991→    if (which === 'widget') return this._dock && this._dock.isWidgetPopoverOpen();
   992→    return false;
   993→  }
   994→
   995→  _open(which, anchor) {
   996→    if (which === 'launcher' && this._launcher) this._launcher.open(anchor);
   997→    else if (which === 'qs' && this._qs) this._qs.open(anchor);
   998→    else if (which === 'cal' && this._cal) this._cal.open(anchor);
   999→    else if (which === 'theme' && this._dock) this._dock.toggleThemePopup();
  1000→    // 'widget' is opened directly by the dock (which emits the bus event); the
  1001→    // shell just closes others via _closeAllExcept.
  1002→  }
  1003→
  1004→  _close(which) {
  1005→    if (which === 'launcher' && this._launcher) this._launcher.close();
  1006→    else if (which === 'qs' && this._qs) this._qs.close();
  1007→    else if (which === 'cal' && this._cal) this._cal.close();
  1008→    else if (which === 'theme' && this._dock) this._dock.closeThemePopup();
  1009→    else if (which === 'widget' && this._dock) this._dock._closeWidgetPopover();
  1010→  }
  1011→
  1012→  disconnectedCallback() {
  1013→    if (this._unsubWindows) this._unsubWindows();
  1014→    if (this._unsubTheme) this._unsubTheme();
  1015→    if (this._unsubBus) this._unsubBus();
  1016→    if (this._resizeHandler) window.removeEventListener('resize', this._resizeHandler);
  1017→    if (this._keyHandler) document.removeEventListener('keydown', this._keyHandler);
  1018→  }
  1019→}
