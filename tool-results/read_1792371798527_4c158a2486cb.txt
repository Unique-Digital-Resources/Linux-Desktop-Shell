  1854→  function makeSlider(sliderEl, fillEl, thumbEl, valueEl, initial, onChange) {
  1855→    let value = initial;
  1856→    let dragging = false;
  1857→    function setVal(v, fire = true) {
  1858→      value = Math.max(0, Math.min(100, Math.round(v)));
  1859→      fillEl.style.width = value + '%';
  1860→      if (thumbEl) thumbEl.style.left = value + '%';
  1861→      if (valueEl) valueEl.textContent = value + '%';
  1862→      sliderEl.setAttribute('aria-valuenow', value);
  1863→      if (fire && onChange) onChange(value);
  1864→    }
  1865→    function fromEvent(clientX) {
  1866→      const r = sliderEl.getBoundingClientRect();
  1867→      const x = Math.max(0, Math.min(r.width, clientX - r.left));
  1868→      setVal((x / r.width) * 100);
  1869→    }
  1870→    setVal(initial, false);
  1871→    sliderEl.addEventListener('mousedown', e => { dragging = true; fromEvent(e.clientX); e.preventDefault(); });
  1872→    sliderEl.addEventListener('touchstart', e => { dragging = true; fromEvent(e.touches[0].clientX); }, { passive: true });
  1873→    document.addEventListener('mousemove', e => { if (dragging) fromEvent(e.clientX); });
  1874→    document.addEventListener('touchmove', e => { if (dragging) fromEvent(e.touches[0].clientX); }, { passive: true });
  1875→    document.addEventListener('mouseup', () => { dragging = false; });
  1876→    document.addEventListener('touchend', () => { dragging = false; });
  1877→    return { setVal, getVal: () => value };
  1878→  }
